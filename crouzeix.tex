\section{Crouzeix's conjecture}
Our goal:
\begin{equation}
    \|f(A)\|_2 \le 2 \sup_{z\in W(A)} |f(z)| 
    \label{C}
\end{equation}

\subsection{Corollaries from \ref{C}}
If $A$, $A_i$ satisfy \ref{C}, then the followings also satisfy \ref{C}: \cite{oloughlin_2023}
\begin{enumerate}
    \item $B = \mu A + \lambda I$ for $\mu,\lambda \in \C$.
    \item $A^*, A^\T$, and the block diagonal matrix $A_1 \oplus \dots \oplus A_n = \diag{A_1,\dots, A_n}$.
    \item The tensor product $N_1 \otimes \dots \otimes A \otimes \dots \otimes N_n $, where each $N_i$ is normal. 
    
    \Prf $N_i = U_i D_i U_i^*$, where $U_i\in U(n)$. Note that $D_i \otimes A = d_{i1}A \oplus d_{i2}A \oplus \dots$. 
    
    Let $N_i \in \C^{n_i \times n_i}, A\in \C^{a\times a}$. Also note that $A \otimes N_i$ is unitarily similar to $N_i \otimes A$ through an order-exchanging unitary linear map $\sigma: x_1 \otimes x_2 \mapsto x_2 \otimes x_1 $, where $x_1 \in \C^{n_i}, x_2 \in \C^a$. So we can permute $A$ to the last. \QED
\end{enumerate}

\begin{equation}
    \psi(A) :=  \sup_{f \in \mathcal{A} (W(A))} \big\{ \|f(A)\|_2 \mid |f(z)| \le 1, z\in W(A) \big\} \le 2

Note that the set of analytic functions $\mathcal{A}(\cdot) \subsetneq C^\infty(\cdot)$.

A sup-attainer function $\hat f_A$ is called an extremal function of $A$. $\hat f$ takes the form $B \circ \varphi$, where $\varphi$ is a conformal mapping from $W(A)^\circ$ to $\D$, and 
\[ B(z) = \exp(i\gamma) \prod_{j=1}^m \frac{z-\alpha_j}{1 - \bar{\alpha_j} z}, \ m \le n-1, \ |\alpha_j|<1 \]

\subsection{What may prove \ref{C}}
\begin{enumerate}
    \item Let $A\in \C^{n\times n}$, $\mathcal{V} \subseteq \C^n$ and $\dim \mathcal{V} \le 2$. Define $A_{\mathcal V} = P_{\mathcal V} A_{| \mathcal V}: \mathcal{V} \to \mathcal{V}$. \cite{oloughlin_2023}
        \subitem If for any polynomial $p$, there exists an above $\mathcal{V}(p)$ such that $\|p(A)\|_2 \le \|p(A_{\mathcal V})\|_2$, then $A$ satisfies \ref{C}. \\
        \Rmk A weaker condition. There exists $\mathcal{V}$ such that $\|\hat f (A)\|_2 \le \|\hat f (A_{\mathcal V})\|_2$. \\
        \Prf $A_{\mathcal{V}}$ is unitarily similar to an $2\times 2$ matrix, so it satisfies \ref{C}. Also, $W(A_{\mathcal{V}}) \subseteq W(A)$. \QED

        \subitem Let $x$ be the principal right singular vector of $\hat f(A)$. If 
        \begin{equation}
            \dim \spn{x, Ax, \dots, A^{n-1}x} \le 2
            \label{dim_Krylov_le_2}
        \end{equation}
        then $A$ satisfies \ref{C}. \\
        \Prf Let $\mathcal{V}$ be the above Krylov subspace. Since $A(\mathcal{V}) \subseteq \mathcal{V}$, we have $\|\hat f (A)\|_2 = \|\hat f (A)_{|\mathcal{V}} \|_2 = \| \hat f (A_\mathcal{V}) \|_2$. \QED 
\end{enumerate}

\subsubsection{Schauderr-Tychonoff fixed point theorem}
Let $K$ be a compact convex subset of a Hausdorff space $X$. Let $f: K \to K$ be continuous. Then $\exists p\in K$ such that $f(p)=p$.


\Prf Suppose not true. $G:=\{ (x, f(x)) \mid x \in K \}$ is compact. Denote the diagonal of $X \times X$ as $\Delta = \{ (x, x) \mid x\in X \}$. 

There exists a convex neighborhood $V$ of $0$, s.t. $(G + V\times V) \cap \Delta = \emptyset$. Also implies $\forall x \in K, f(x) \notin x+V$. 

Define Minkowski functional of V: $\mu_V(x):= \inf \{t>0 \mid t^{-1}x \in V, x\in K\}$. Note that $\mu_V(x) < 1 \Leftrightarrow x\in V$. 

Define $\alpha(x) := \max (0, 1-\mu_V(x))$. Choose $x_1,\dots, x_n$ s.t. $\{x_i+V\}$ covers $K$. Define $\alpha_i(x):= \alpha(x-x_i)$ and $\beta_i(x) := \frac{\alpha_i(x)}{\sum_{i=1}^n \alpha_i(x)}$. 

Define $H := \co \{x_1,\dots,x_n\}$. Define $g(x):=\sum_{i=1}^n \beta_i(x)x_i$. Then $g\circ f : H \subset K \to K \to H$. Since $H$ is finite-dim simplex, by Brouwer's fixed point thm, exists $x^* \in H$ s.t. $g(f(x^*)) = x^*$.

$\beta_i(x)=0 \Leftrightarrow x - x_i \notin V$ implies $\forall x\in K, x-g(x) = \sum_{i=1}^n \beta_i(x)(x-x_i) \in V$. Also true for $x=f(x^*)$, so $f(x^*) - x^* \in V$, contradiction. \QED
